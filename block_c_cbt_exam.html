import java.net.URLEncoder; import java.nio.charset.StandardCharsets; import java.util.*; import java.util.stream.Collectors; public class ExamManager { private static final String KAKAO_CHATBOT_URL = "https://pf.kakao.com/_xxxxxx/chat"; private final List masterQuestionBank; // 전체 문제 DB public ExamManager(List masterQuestionBank) { this.masterQuestionBank = masterQuestionBank; } // ========================================== // [통합 1] 문제(Question) 내부 클래스 // ========================================== public static class Question { private Long id; private String licenseName; // 자격증 종류 (예: 3급 항해사, 3급 기관사 등) private String subject; // 세부 과목 private String topic; // 단원/주제 private String questionText; private List options; // 보기 4개 private int answer; // 정답 번호 (1~4) private String explanation; public Question(Long id, String licenseName, String subject, String topic, String questionText, List options, int answer, String explanation) { this.id = id; this.licenseName = licenseName; this.subject = subject; this.topic = topic; this.questionText = questionText; this.options = new ArrayList<>(options); this.answer = answer; this.explanation = explanation; } // 보기(Option) 및 정답 번호 무작위 셔플 public void shuffleOptions() { String correctAnswerText = options.get(answer - 1); Collections.shuffle(options); // 섞인 리스트에서 정답 텍스트의 새로운 위치(1~4) 재설정 this.answer = options.indexOf(correctAnswerText) + 1; } // Getters public Long getId() { return id; } public String getLicenseName() { return licenseName; } public String getSubject() { return subject; } public String getTopic() { return topic; } public String getQuestionText() { return questionText; } public List getOptions() { return options; } public int getAnswer() { return answer; } public String getExplanation() { return explanation; } } // ========================================== // [통합 2] 자격증 추출 & 30문항 랜덤 로직 // ========================================== /** * 1. DB에 존재하는 자격증 목록 조회 */ public List getAvailableLicenses() { return masterQuestionBank.stream() .map(Question::getLicenseName) .distinct() .sorted() .collect(Collectors.toList()); } /** * 2. 선택한 '단일 자격증'에서만 무작위 30문제 추출 (보기 셔플 포함) */ public List createRandomExam(String selectedLicense, int targetQuestionCount) { // 선택한 자격증의 문제만 필터링 (종합/혼합 제거) List filteredQuestions = masterQuestionBank.stream() .filter(q -> q.getLicenseName().equalsIgnoreCase(selectedLicense)) .collect(Collectors.toList()); if (filteredQuestions.isEmpty()) { return Collections.emptyList(); } // 문항 순서 무작위 셔플 Collections.shuffle(filteredQuestions); // 지정한 개수(30개) 만큼 추출 int count = Math.min(targetQuestionCount, filteredQuestions.size()); List examQuestions = new ArrayList<>(filteredQuestions.subList(0, count)); // 각 문제별 1~4번 보기 순서 셔플 for (Question q : examQuestions) { q.shuffleOptions(); } return examQuestions; } /** * 3. 디디쌤 카카오 챗봇 오답 질의 링크 생성 */ public String generateKakaoChatbotLink(Question question) { String prompt = String.format("디디쌤, [%s-%s] 단원 문제 '%s...' 오답 개념 및 보완 문제 알려줘!", question.getLicenseName(), question.getTopic(), question.getQuestionText().length() > 20 ? question.getQuestionText().substring(0, 20) : question.getQuestionText() ); String encodedPrompt = URLEncoder.encode(prompt, StandardCharsets.UTF_8); return KAKAO_CHATBOT_URL + "?input=" + encodedPrompt; } }
